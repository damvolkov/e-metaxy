import sys
import numpy as np
from scipy.stats import bootstrap
for name in sys.argv[1:]:
    rows = [r.split("\t") for r in open(f"/tmp/bench-cutout/out/{name}.tsv").read().split("\n") if r]
    d = {}
    for img, arm, ms, f, iou in rows:
        d.setdefault(arm, {})[img] = (float(ms), float(f), float(iou))
    imgs = sorted(d["a2"])
    for j, metric in ((1, "F"), (2, "IoU")):
        diff = np.array([d["l"][i][j] - d["a2"][i][j] for i in imgs])
        ci = bootstrap((diff,), np.mean, method="BCa", confidence_level=0.95, n_resamples=20000, random_state=1).confidence_interval
        print(f"{name} {metric}: mean diff {diff.mean():+.4f} CI [{ci.low:+.4f}, {ci.high:+.4f}] L wins {(diff>0).sum()}/{len(diff)} loses {(diff<0).sum()}")
    for arm in ("a2", "l"):
        ms = np.array([d[arm][i][0] for i in imgs]); f = np.array([d[arm][i][1] for i in imgs]); iou = np.array([d[arm][i][2] for i in imgs])
        print(f"   {arm}: F med {np.median(f):.3f} IoU med {np.median(iou):.3f} ms p50 {np.median(ms):.1f} p95 {np.percentile(ms,95):.1f}")
