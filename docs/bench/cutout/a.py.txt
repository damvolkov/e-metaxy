"""Arm A: flat-field by closing, Sauvola, darkness gate, specks dropped. Parameters fixed on the dev set only."""
import sys, time
import cv2
import numpy as np
from skimage.filters import threshold_sauvola

WINDOW, K, GATE, SPECK, CLOSE = 25, 0.2, 150, 30, 31


def cut(img: np.ndarray) -> np.ndarray:
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    background = cv2.morphologyEx(gray, cv2.MORPH_CLOSE, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (CLOSE, CLOSE)))
    flat = np.clip(gray.astype(np.float32) / np.maximum(background, 1) * 255, 0, 255)
    ink = (flat < threshold_sauvola(flat, window_size=WINDOW, k=K)) & (flat < GATE)
    count, labels, stats, _ = cv2.connectedComponentsWithStats(ink.astype(np.uint8), connectivity=8)
    keep = np.zeros(count, bool)
    keep[1:] = stats[1:, cv2.CC_STAT_AREA] >= SPECK
    return (keep[labels] * 255).astype(np.uint8)


for line in sys.stdin:
    src, dst = line.split()
    img = cv2.imread(src)
    t = time.perf_counter()
    ink = cut(img)
    ms = (time.perf_counter() - t) * 1000
    cv2.imwrite(dst, ink)
    print(f"{ms:.3f}", flush=True)
