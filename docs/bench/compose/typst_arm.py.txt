"""Arm TY: typst (Rust, PyPI). stdin: `data.json out.pdf` → stdout: ms, bytes."""
import sys, time
import typst
TEMPLATE = open("/tmp/bench-compose/scripts/invoice.typ", "rb").read()
for line in sys.stdin:
    src, dst = line.split()
    data = open(src).read()
    t = time.perf_counter()
    pdf = typst.compile(TEMPLATE, sys_inputs={"data": data}, pdf_standards="a-2b", timestamp=0, ignore_system_fonts=True)
    ms = (time.perf_counter() - t) * 1000
    open(dst, "wb").write(pdf)
    print(f"{ms:.3f} {len(pdf)}", flush=True)
