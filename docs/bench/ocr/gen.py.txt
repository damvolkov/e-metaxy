"""Synthetic scanned pages with known words and boxes, ES and EN, three degradation levels.
Usage: gen.py OUT_DIR PAGES_PER_LANG SEED → NNN-<lang>-<level>.jpg + .json {words: [{text, box: [x0,y0,x1,y1]}]} (reading order)."""
import json, sys
from pathlib import Path
import cv2
import numpy as np
from PIL import Image, ImageDraw, ImageFont

W, H = 1240, 1754
FONTS = ["/usr/share/fonts/truetype/liberation/LiberationSerif-Regular.ttf",
         "/usr/share/fonts/truetype/liberation/LiberationSans-Regular.ttf",
         "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"]
CORPUS = {

 "en": ["Invoice number 42 from Northern Steel Ltd. amounts to 1,250.00 euros, taxes included.",
        "The contractor agrees to deliver the works before 15 March 2027 under the conditions agreed.",

        "Attached is the delivery note with reference EN-2027-0318 and the breakdown of the units supplied.",

        "The shipping address is 25 High Street, Flat 3B, London EC1A 1BB, and the phone number 020 7946 0958."],
}
LEVELS = ("clean", "noisy", "bad")


def page(lang: str, rng: np.random.Generator) -> tuple[Image.Image, list[dict]]:
    font = ImageFont.truetype(FONTS[rng.integers(0, len(FONTS))], int(rng.integers(24, 31)))
    img = Image.new("RGB", (W, H), "white")
    draw = ImageDraw.Draw(img)
    sentences = [CORPUS[lang][i] for i in rng.permutation(len(CORPUS[lang]))[: rng.integers(7, 11)]]
    words, y, x0 = [], 110, 100
    lead = int(font.size * 1.55)
    for para in sentences:
        x = x0
        for w in para.split():
            width = draw.textlength(w + " ", font=font)
            if x + width > W - 100:
                x, y = x0, y + lead
            l, t, r, b = draw.textbbox((x, y), w, font=font)
            draw.text((x, y), w, font=font, fill=(20, 20, 20))
            words.append({"text": w, "box": [l, t, r, b]})
            x += width
        y += int(lead * 1.7)
    return img, words


def degrade(img: Image.Image, words: list[dict], level: str, rng: np.random.Generator):
    a = np.asarray(img).astype(np.float32)
    angle = {"clean": 0.0, "noisy": rng.uniform(-1, 1), "bad": rng.uniform(-2.5, 2.5)}[level]
    m = cv2.getRotationMatrix2D((W / 2, H / 2), angle, 1.0)
    a = cv2.warpAffine(a, m, (W, H), borderValue=(255, 255, 255))
    out = []
    for w in words:
        l, t, r, b = w["box"]
        pts = cv2.transform(np.float32([[[l, t], [r, t], [r, b], [l, b]]]), m)[0]
        out.append({"text": w["text"], "box": [float(pts[:, 0].min()), float(pts[:, 1].min()), float(pts[:, 0].max()), float(pts[:, 1].max())]})
    sigma = {"clean": 2, "noisy": 10, "bad": 18}[level]
    blur = {"clean": 0.0, "noisy": 0.8, "bad": 1.3}[level]
    a = cv2.GaussianBlur(a, (0, 0), blur) if blur else a
    if level == "bad":
        a = cv2.resize(cv2.resize(a, (int(W * 0.6), int(H * 0.6)), interpolation=cv2.INTER_AREA), (W, H), interpolation=cv2.INTER_LINEAR)
        gx, gy = np.meshgrid(np.linspace(0, 1, W), np.linspace(0, 1, H))
        a = 255 - (255 - a) * 0.75
        a = a * (1 - 0.3 * (0.7 * gx + 0.3 * gy))[..., None]
    a = np.clip(a + rng.normal(0, sigma, a.shape), 0, 255).astype(np.uint8)
    q = {"clean": 90, "noisy": 60, "bad": 35}[level]
    ok, jpg = cv2.imencode(".jpg", a[..., ::-1], [cv2.IMWRITE_JPEG_QUALITY, q])
    return jpg.tobytes(), out


def main() -> None:
    out, n, seed = Path(sys.argv[1]), int(sys.argv[2]), int(sys.argv[3])
    rng = np.random.default_rng(seed)
    out.mkdir(parents=True, exist_ok=True)
    for i in range(n):
        for lang in CORPUS:
            img, words = page(lang, rng)
            for level in LEVELS:
                jpg, boxes = degrade(img, words, level, rng)
                stem = f"{i:03d}-{lang}-{level}"
                (out / f"{stem}.jpg").write_bytes(jpg)
                (out / f"{stem}.json").write_text(json.dumps({"lang": lang, "level": level, "words": boxes}))


main()
